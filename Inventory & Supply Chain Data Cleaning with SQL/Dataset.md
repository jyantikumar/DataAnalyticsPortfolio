CREATE DATABASE inventory_supply_chain;
USE inventory_supply_chain;

CREATE TABLE suppliers (
    supplier_id INT,
    supplier_name VARCHAR(100),
    contact_name VARCHAR(100),
    email VARCHAR(100),
    phone VARCHAR(30),
    city VARCHAR(100),
    supplier_status VARCHAR(30)
);

CREATE TABLE warehouses (
    warehouse_id INT,
    warehouse_name VARCHAR(100),
    city VARCHAR(100),
    warehouse_manager VARCHAR(100)
);

CREATE TABLE products (
    product_id INT,
    supplier_id INT,
    product_name VARCHAR(100),
    category VARCHAR(50),
    unit_cost VARCHAR(30),
    selling_price VARCHAR(30),
    reorder_level VARCHAR(20),
    product_status VARCHAR(30)
);

CREATE TABLE inventory (
    inventory_id INT,
    warehouse_id INT,
    product_id INT,
    stock_quantity VARCHAR(20),
    last_stock_update VARCHAR(30)
);

CREATE TABLE customers (
    customer_id INT,
    first_name VARCHAR(50),
    last_name VARCHAR(50),
    email VARCHAR(100),
    phone VARCHAR(30),
    city VARCHAR(100),
    customer_type VARCHAR(30),
    signup_date VARCHAR(30)
);

CREATE TABLE sales (
    sale_id INT,
    sale_date VARCHAR(30),
    customer_id INT,
    product_id INT,
    warehouse_id INT,
    quantity VARCHAR(20),
    payment_method VARCHAR(30),
    sale_status VARCHAR(30),
    sales_rep VARCHAR(100)
);
